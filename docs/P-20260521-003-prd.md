# PRD — cultivation-simulator V48 插件市场系统

## 1. 概述

**项目**: cultivation-simulator (修仙模拟器)
**版本**: V48
**提案ID**: P-20260521-003
**方向**: E (插件市场系统)
**设计借鉴**: ruflo-design Plugin架构 + nanobot-design Tool注册机制

### 1.1 功能目标

实现玩家可创建、上传、下载自定义修仙模块的插件市场系统。借鉴ruflo的definePlugin模式，定义标准化的插件接口，支持技能包、资源包、剧情包、界面主题等多种插件类型。

### 1.2 核心价值

- 玩家可创建自定义修仙模块（技能、资源、剧情、主题）
- 标准化插件注册机制，隔离执行环境
- 内置插件市场UI，支持浏览/安装/评分/评论
- 插件依赖声明和版本校验

```javascript
function renderPluginMarket() {
  let html = '<div class="panel">';
  html += '<div class="panel-header">';
  html += '<span class="panel-title">📦 插件市场</span>';
  html += '</div>';

  // 分类Tab

  html += '</div>';

  html += '</div>';

// 插件卡片
function renderMarketPluginCard(plugin) {
  const isInstalled = gameState.plugins.installed[plugin.id];
  const isFavorite = gameState.plugins.favorites.includes(plugin.id);
  
  let html = `<div class="plugin-card">`;
  html += `<div class="plugin-header">`;
  html += `<span class="plugin-icon">${plugin.icon}</span>`;
  html += `<span class="plugin-name">${plugin.name}</span>`;
  html += `<span class="plugin-version">v${plugin.version}</span>`;
  html += `</div>`;
  html += `<div class="plugin-desc">${plugin.description}</div>`;
  html += `<div class="plugin-meta">`;
  html += `<span>👤 ${plugin.author}</span>`;
  html += `<span>📥 ${plugin.installCount}</span>`;
  html += `<span>⭐ ${plugin.rating.toFixed(1)}</span>`;
  html += `</div>`;
  html += `<div class="plugin-actions">`;
  if (isInstalled) {
    html += `<button class="btn btn-installed" onclick="openPluginDetail('${plugin.id}')">已安装</button>`;
  } else {
    html += `<button class="btn btn-install" onclick="installPlugin('${plugin.id}')">安装</button>`;
  }
  html += `<button class="btn btn-fav ${isFavorite ? 'active' : ''}" onclick="toggleFavorite('${plugin.id}')">❤️</button>`;
  html += `</div>`;
  html += `</div>`;
  return html;
}
```

```javascript
function openPluginDetail(pluginId) {

        </div>
      </div>
      <div class="detail-desc">${plugin.description}</div>
      <div class="detail-category">分类: ${PLUGIN_CATEGORIES[plugin.category].icon} ${PLUGIN_CATEGORIES[plugin.category].label}</div>
      ${plugin.dependencies.length > 0 ? `<div class="detail-deps">依赖: ${plugin.dependencies.join(', ')}</div>` : ''}
      <div class="detail-actions">
        ${gameState.plugins.installed[plugin.id] 
          ? `<button class="btn btn-uninstall" onclick="uninstallPlugin('${plugin.id}')">卸载</button>
             <button class="btn btn-toggle ${gameState.plugins.enabled.includes(plugin.id) ? 'btn-disable' : 'btn-enable'}" 
                     onclick="togglePlugin('${plugin.id}')">${gameState.plugins.enabled.includes(plugin.id) ? '禁用' : '启用'}</button>`
          : `<button class="btn btn-install" onclick="installPlugin('${plugin.id}')">安装</button>`
        }
      </div>
    </div>
  `);
}
```

```javascript
};

// 钩子调用系统
function callPluginHook(hookName, ...args) {
  let result = args;
  gameState.plugins.enabled.forEach(pluginId => {
    const plugin = gameState.plugins.installed[pluginId];
    if (plugin && plugin.hooks && plugin.hooks[hookName]) {
      try {
        result = plugin.hooks[hookName](...result) || result;
      } catch (e) {
        console.warn(`Plugin ${pluginId} hook ${hookName} error:`, e);
      }
    }
  });
  return result;
}
```

### 3.5 插件安装/卸载逻辑

```javascript
function installPlugin(pluginId) {
  const marketPlugin = BUILT_IN_PLUGINS.find(p => p.id === pluginId);
  if (!marketPlugin) return;

  // 检查依赖
  for (const dep of marketPlugin.dependencies) {
    if (!gameState.plugins.installed[dep]) {
      showToast(`缺少依赖插件: ${dep}`);
      return;
    }
  }

  // 安装
  gameState.plugins.installed[pluginId] = { ...marketPlugin, enabled: true };
  if (!gameState.plugins.enabled.includes(pluginId)) {
    gameState.plugins.enabled.push(pluginId);
  }

  // 调用onSpawn
  if (marketPlugin.hooks && marketPlugin.hooks.onSpawn) {
    try { marketPlugin.hooks.onSpawn(); } catch(e) {}
  }

  marketPlugin.installCount++;
  showToast(`已安装 ${marketPlugin.name}`);
  updateDisplay();
}

function uninstallPlugin(pluginId) {
  const plugin = gameState.plugins.installed[pluginId];
  if (!plugin) return;

  // 检查是否有插件依赖此插件
  Object.values(gameState.plugins.installed).forEach(p => {
    if (p.dependencies.includes(pluginId)) {
      showToast(`无法卸载: ${p.name} 依赖此插件`);
      return;
    }
  });

  // 调用onDestroy
  if (plugin.hooks && plugin.hooks.onDestroy) {
    try { plugin.hooks.onDestroy(); } catch(e) {}
  }

  delete gameState.plugins.installed[pluginId];
  gameState.plugins.enabled = gameState.plugins.enabled.filter(id => id !== pluginId);
  showToast(`已卸载 ${plugin.name}`);
  updateDisplay();
}

function togglePlugin(pluginId) {
  const idx = gameState.plugins.enabled.indexOf(pluginId);
  if (idx >= 0) {
    gameState.plugins.enabled.splice(idx, 1);
    showToast('插件已禁用');
  } else {
    gameState.plugins.enabled.push(pluginId);
    const plugin = gameState.plugins.installed[pluginId];
    if (plugin && plugin.hooks && plugin.hooks.onSpawn) {
      try { plugin.hooks.onSpawn(); } catch(e) {}
    }
    showToast('插件已启用');
  }
  updateDisplay();
}
```
